import { useState } from 'react';
import { sevenDimensions } from './kitchenTable7D';

export default function SevenDPanel({ mode = 'working', onFireSevenGen }: { mode?: string; onFireSevenGen?: () => void }) {
  const [checks, setChecks] = useState<Record<string, boolean>>({});

  const toggle = (id: string) => setChecks(prev => ({ ...prev, [id]: !prev[id] }));

  const isCompact = mode === 'sounding';
  const isReview = mode === 'review';

  return (
    <div className="border-l border-green-900/30 bg-[#1a2f24] p-4 overflow-auto">
      <h3 className="text-green-300 font-medium mb-4 flex items-center gap-2">
        7D Filter <span className="text-xs opacity-60">(North Star Guided)</span>
      </h3>

      <div className={isCompact ? "grid grid-cols-1 gap-3" : "space-y-6"}>
        {sevenDimensions.map(dim => (
          <div key={dim.id} className="border border-green-900/20 rounded p-3 bg-[#13221a]">
            <label className="flex items-start gap-3 cursor-pointer">
              <input
                type="checkbox"
                checked={checks[dim.id] || false}
                onChange={() => toggle(dim.id)}
                className="mt-1 accent-green-600"
              />
              <div className="flex-1">
                <div className="flex items-center gap-2 text-green-200">
                  <span>{dim.symbol}</span>
                  <span className="font-medium">{dim.name}</span>
                </div>
                <p className="text-sm text-green-400 mt-1">{dim.prompt}</p>
                <p className="text-xs text-green-500 mt-2">{dim.check}</p>
                {dim.gord && <p className="text-xs italic text-amber-300 mt-1">Gord: {dim.gord}</p>}
              </div>
            </label>

            {dim.id === 'future' && isReview && (
              <button
                onClick={onFireSevenGen}
                className="mt-3 text-xs bg-green-900 hover:bg-green-800 px-3 py-1 rounded"
              >
                Run Ishmael Seven-Gen Test
              </button>
            )}
          </div>
        ))}
      </div>

      <div className="mt-6 pt-4 border-t border-green-900/30">
        <button className="w-full bg-amber-900 hover:bg-amber-800 text-amber-100 py-3 rounded font-medium">
          Does this pass the Gord Test at this kitchen table?
        </button>
      </div>
    </div>
  );
}
